{{- /* Renders Markdown links. This is Hugo's embedded link render hook (v0.167.0, tpl/tplimpl/embedded/templates/_markup/render-link.html), which resolves a link to the page or resource it names, plus two additions. First, a link to a section's landing page goes to the section instead, where that page is published; its own URL redirects there. That includes a link to a heading on the same page. See partials/page-link-target.html. Second, a root path that doesn't resolve, such as an alias path or a static file, gets the site's base path, so it keeps working when the docs are served under a prefix such as www.chainguard.dev/docs/. With a base URL at the root of its host it changes nothing. See partials/site-path.html. */ -}} {{- $u := urls.Parse .Destination -}} {{- $href := $u.String -}} {{- if strings.HasPrefix $u.String "#" -}} {{- $href = printf "%s#%s" (partial "page-link-target.html" .PageInner).RelPermalink $u.Fragment -}} {{- else if and $href (not $u.IsAbs) -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or ($.PageInner.GetPage $path) ($.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $href = .RelPermalink -}} {{- with $.PageInner.GetPage $path -}} {{- $href = (partial "page-link-target.html" .).RelPermalink -}} {{- end -}} {{- with $u.RawQuery -}} {{- $href = printf "%s?%s" $href . -}} {{- end -}} {{- with $u.Fragment -}} {{- $href = printf "%s#%s" $href . -}} {{- end -}} {{- end -}} {{- end -}} {{- $href = partial "site-path.html" $href -}} {{ .Text }} {{- /**/ -}}
